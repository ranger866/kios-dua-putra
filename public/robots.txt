# Kios Pro Robots Exclusion File

# Larang bot mengindeks halaman admin sensitif
Disallow: /admin.html
Disallow: /api/

# Larang bot mengindeks file laporan atau database (jika terekspos)
Disallow: /data/

# Izinkan bot hanya melihat halaman login (untuk branding saja)
Allow: /login.html
Allow: /index.html

# Beritahu mesin pencari untuk tidak menyimpan cache halaman admin
Noindex: /admin.html
